\begin{frame}{\small ONERA M6 -- Modeling Choices Across Submissions}
\scriptsize

\begin{itemize}

    \item \textbf{Turbulence:}
    $k$--$\omega$ formulations were dominant (\textbf{80\%}), with SA used by \textbf{20\%}.

    \vspace{0.10cm}

    \item \textbf{Droplet formulation:}
    Predominantly Eulerian (\textbf{80\%}), with \textbf{20\%} Lagrangian.

    \vspace{0.10cm}

    \item \textbf{Heat transfer:}
    Documented approaches included recovery-temperature and boundary-layer integral methods.

    \vspace{0.10cm}

    \item \textbf{Thermodynamics:}
    Messinger-based models were dominant (\textbf{80\%}), with a SWIM-based approach also represented.

    \vspace{0.10cm}

    \item \textbf{Geometry evolution:}
    Most commonly based on Lagrangian/IB/node displacement; level-set and interpolation-based approaches were also reported.

    \vspace{0.10cm}

    \item \textbf{Ice accretion strategy:}
    Both single-shot (\textbf{4 submissions}) and multilayer/multishot (\textbf{3 submissions}) approaches were represented.

\end{itemize}
